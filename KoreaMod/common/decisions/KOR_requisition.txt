	# The Philippines asking Washington for rifles, transplanted. Loyalty
	# raises the odds; three refusals in a row cost you autonomy.
	KOR_requisition_infantry = {
		icon = generic_operation
		allowed = { original_tag = KOR }
		available = { is_subject = yes }

					modifier = { factor = 1.5 has_country_flag = KOR_naisen_ittai_committed }

					add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 2500 producer = JAP }
					clr_country_flag = KOR_requisition_refused_1
					clr_country_flag = KOR_requisition_refused_2
				}
				45 = {
					custom_effect_tooltip = KOR_requisition_refused_tt
					if = {
						limit = { has_country_flag = KOR_requisition_refused_2 }
						add_autonomy_ratio = { value = -0.05 }
						clr_country_flag = KOR_requisition_refused_1
						clr_country_flag = KOR_requisition_refused_2
					}
					else_if = {
						limit = { has_country_flag = KOR_requisition_refused_1 }
						set_country_flag = KOR_requisition_refused_2
					}
					else = { set_country_flag = KOR_requisition_refused_1 }
				}
			}
		}
