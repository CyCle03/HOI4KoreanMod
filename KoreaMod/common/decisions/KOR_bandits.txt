KOR_bandit_category = {

	# The same raids across the Manchurian border are a nuisance to one branch
	# and a recruiting pool to another. One event, two meanings.
	#
	# Both answers are open from January 1936, because the raids start then
	# and the flag-defacement incident that assigns you a branch does not
	# happen until August. Gating the responses behind branch flags left the
	# first eight months with nothing to do but absorb the damage.

	KOR_bandit_suppression = {
		icon = generic_army_support
		allowed = { original_tag = KOR }
		visible = {
			OR = { controls_state = 527 controls_state = 1028 }
			NOT = { has_country_flag = KOR_kil_path }
		}
		cost = 0
		custom_cost_trigger = { command_power > 9 }
		custom_cost_text = KOR_bandit_cost_tt
		days_re_enable = 60

		ai_will_do = { factor = 15 }

		complete_effect = {
			add_command_power = -10
			527 = { add_resistance = -15 }
			1028 = { add_resistance = -15 }
			add_stability = 0.02
			add_manpower = -1500
			clr_country_flag = KOR_bandits_unchecked
			# 130, not 120: the reprieve has to outlast the mission cycle it is
			# meant to cover, or the timeout lands in the gap and the decision
			# does nothing at all - which is what it did before.
			set_country_flag = { flag = KOR_bandits_handled days = 130 }
		}
	}

	KOR_bandit_recruit = {
		icon = generic_army_support
		allowed = { original_tag = KOR }
		visible = {
			NOT = { has_country_flag = KOR_naisen_ittai_committed }
		}
		cost = 15
		days_re_enable = 60

		ai_will_do = { factor = 12 }

		complete_effect = {
			add_war_support = 0.02
			add_manpower = 3000
			army_experience = 5
			if = {
				limit = { country_exists = JAP NOT = { tag = JAP } }
				JAP = { add_opinion_modifier = { target = ROOT modifier = KOR_trusteeship_resentment } }
			}
		}
	}

	# If nobody deals with them, the damage accumulates.
	KOR_bandit_neglect = {
		icon = generic_army_support
		allowed = { original_tag = KOR }
		visible = { OR = { controls_state = 527 controls_state = 1028 } }
		available = { always = no }
		fire_only_once = no
		days_mission_timeout = 120
		activation = { OR = { controls_state = 527 controls_state = 1028 } }
		is_good = no

		timeout_effect = {
			# Only the ones nobody answered do damage. Suppression sets
			# KOR_bandits_handled for 130 days and this reads it; before, the
			# decision cleared a flag that nothing anywhere read, so paying ten
			# command power bought exactly nothing.
			if = {
				limit = { NOT = { has_country_flag = KOR_bandits_handled } }
				random_state = {
					limit = { OR = { state = 527 state = 1028 } CONTROLLER = { tag = ROOT } }
					damage_building = { type = infrastructure damage = 1 }
				}
				add_political_power = -15
				set_country_flag = KOR_bandits_unchecked
			}
		}
	}
}
