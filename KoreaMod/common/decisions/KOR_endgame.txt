KOR_endgame_category = {

	# For most countries the war ending is the ending. Here it is where the
	# game restarts. One number - the recognition gauge - decides whether you
	# come out of it a sovereign state or a zone under military government.

	KOR_show_recognition = {
		icon = generic_political_actions
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		available = { always = no }
		visible = { has_country_flag = KOR_underground_built }
		fire_only_once = no
		cost = 0
		modifier = { political_power_gain = 0 }
	}

	# --- domestic entry ---------------------------------------------------

	KOR_homeland_entry = {
		icon = generic_operation
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		visible = { has_country_flag = KOR_eagle_ready }
		available = {
			has_country_flag = KOR_eagle_ready
			OR = {
				JAP = { surrender_progress > 0.7 }
				JAP = { has_capitulated = yes }
			}
		}
		cost = 75
		fire_only_once = yes
		days_remove = 30

		ai_will_do = { factor = 30 }

		remove_effect = {
			random_list = {
				65 = {
					modifier = { factor = 1.4 has_completed_focus = KOR_military_academy }
					525 = { set_state_controller = ROOT }
					527 = { set_state_controller = ROOT }
					set_variable = { KOR_recognition_delta = 25 }
					KOR_lift_recognition_cap = yes
					KOR_add_recognition = yes
					custom_effect_tooltip = KOR_homeland_entry_success_tt
				}
				35 = {
					custom_effect_tooltip = KOR_homeland_entry_failure_tt
					add_timed_idea = { idea = KOR_failed_entry days = 90 }
					add_manpower = -20000
				}
			}
		}
	}

	# --- partition --------------------------------------------------------

	KOR_38th_parallel = {
		icon = generic_army_support
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		visible = {
			OR = { JAP = { has_capitulated = yes } has_global_flag = KOR_japan_collapsing }
			check_variable = { KOR_recognition < 60 }
		}
		available = {
			SOV = { has_war_with = JAP }
			check_variable = { KOR_recognition < 60 }
		}
		cost = 0
		fire_only_once = yes
		is_good = no

		complete_effect = {
			if = {
				limit = { has_country_flag = KOR_kil_path }
				# The League loses the south instead.
				525 = { set_state_controller = USA }
				custom_effect_tooltip = KOR_partition_south_tt
			}
			else = {
				527 = { set_state_controller = SOV }
				custom_effect_tooltip = KOR_partition_north_tt
			}
			set_country_flag = KOR_divided_state
			add_stability = -0.10
		}
	}

	# --- trusteeship ------------------------------------------------------

	KOR_accept_trusteeship = {
		icon = generic_political_actions
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		visible = {
			has_country_flag = KOR_cairo_promise
			check_variable = { KOR_recognition > 39 }
			check_variable = { KOR_recognition < 71 }
		}
		available = { has_country_flag = KOR_cairo_promise }
		cost = 25
		fire_only_once = yes

		ai_will_do = { factor = 20 }

		complete_effect = {
			set_country_flag = KOR_trusteeship_declared
			if = { limit = { is_subject = yes } add_autonomy_ratio = { value = -0.30 } }
			custom_effect_tooltip = KOR_trusteeship_accept_tt
			add_timed_idea = { idea = KOR_trusteeship days = 1825 }
		}
	}

	KOR_reject_trusteeship = {
		icon = generic_army_support
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		visible = {
			has_country_flag = KOR_cairo_promise
			check_variable = { KOR_recognition > 39 }
			check_variable = { KOR_recognition < 71 }
		}
		available = { has_country_flag = KOR_cairo_promise }
		cost = 50
		fire_only_once = yes

		ai_will_do = { factor = 10 }

		complete_effect = {
			set_country_flag = KOR_trusteeship_rejected
			add_war_support = 0.15
			every_owned_state = { add_resistance = 30 }
			random_list = {
				60 = { custom_effect_tooltip = KOR_civil_war_avoided_tt add_stability = -0.10 }
				40 = { custom_effect_tooltip = KOR_civil_war_roll_tt add_stability = -0.25 set_country_flag = KOR_civil_war_risk }
			}
		}
	}

	# --- the asset race ---------------------------------------------------

	KOR_asset_race = {
		icon = generic_industry
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		visible = { JAP = { surrender_progress > 0.5 } }
		available = { always = no }
		activation = { JAP = { surrender_progress > 0.5 } }
		days_mission_timeout = 180
		is_good = no
		fire_only_once = yes

		modifier = { industrial_capacity_factory = -0.05 }

		timeout_effect = {
			if = {
				limit = { has_country_flag = KOR_governorate_disarmed }
				custom_effect_tooltip = KOR_asset_race_won_tt
			}
			else = {
				add_ideas = KOR_stripped_industry
				custom_effect_tooltip = KOR_asset_race_lost_tt
			}
		}
	}

	# --- north-south unity talks ------------------------------------------

	KOR_unification_talks = {
		icon = generic_political_actions
		allowed = { OR = { original_tag = KOR original_tag = CHO } }
		visible = {
			has_country_flag = KOR_divided_state
			OR = { has_country_flag = KOR_kpg_path has_country_flag = KOR_kpr_path }
		}
		available = {
			check_variable = { KOR_scale_right > 39 }
			check_variable = { KOR_scale_right < 61 }
		}
		cost = 100
		days_remove = 90
		fire_only_once = yes

		ai_will_do = { factor = 20 }

		remove_effect = {
			random_list = {
				50 = {
					modifier = { factor = 2 has_completed_focus = KPR_coalition_line }
					add_stability = 0.20
					clr_country_flag = KOR_divided_state
					every_state = { limit = { is_core_of = ROOT } transfer_state_to = ROOT }
					custom_effect_tooltip = KOR_unification_success_tt
				}
				50 = {
					add_stability = -0.20
					set_country_flag = KOR_civil_war_risk
					custom_effect_tooltip = KOR_unification_failure_tt
				}
			}
		}
	}
}
