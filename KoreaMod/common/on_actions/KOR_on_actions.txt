on_actions = {

	on_startup = {
		effect = {
			# --- Korea's starting condition. Applied to both tags, because
			# --- either one can be the tag the player is holding: KOR when
			# --- the peninsula is only releasable, CHO when it is split off.
			# --- Everything the tree and the decisions read has to exist on
			# --- whichever one you are actually playing.
			KOR = { KOR_common_setup = yes }
			CHO = { KOR_common_setup = yes }

			# --- Governorate mode. The peninsula already belongs to CHO -
			# --- that is declared in the six history/states files, not here,
			# --- because the country selection screen reads history and
			# --- nothing else. Doing it in on_startup meant Korea was
			# --- Japanese while you were choosing and then broke away the
			# --- instant the game began, and CHO was not selectable at all.
			# --- What is left for startup is the diplomacy, which has no
			# --- map to redraw.
			if = {
				limit = { has_game_rule = { rule = KOR_start_mode option = GOVERNORATE } }

				CHO = {
					add_ideas = KOR_governorate_police
					set_cosmetic_tag = KOR_jap
				}

				JAP = {
					set_autonomy = { target = CHO autonomy_state = autonomy_integrated_puppet }
					# The metropole loses direct use of Korean manpower and
					# industry; the integrated-puppet share plus this spirit
					# is what puts it back.
					add_ideas = JAP_korean_integration
				}

				set_global_flag = KOR_governorate_active
			}

			# --- Releasable-only mode. Hand the peninsula back and leave the
			# --- 1936 map exactly as the base game draws it. Japan's AI then
			# --- sees nothing different, which is the whole point of this
			# --- option; the cost is that Korea cannot be picked at start.
			else = {
				KOR_return_peninsula_to_japan = yes
			}
		}
	}

	# The flag-defacement incident, and Cairo, are fired from here so that
	# they work whichever tag the player is holding.
	on_daily_KOR = {
		effect = {
			if = {
				limit = {
					date > "1936.8.24"
					NOT = { has_country_flag = KOR_trigger_event_fired }
					OR = { is_subject = yes owns_state = 525 has_country_flag = KOR_underground_built }
				}
				set_country_flag = KOR_trigger_event_fired
				country_event = { id = korea_triggers.1 days = 1 }
			}
			if = {
				limit = {
					date > "1943.11.30"
					NOT = { has_country_flag = KOR_cairo_fired }
				}
				set_country_flag = KOR_cairo_fired
				country_event = { id = korea_triggers.2 days = 1 }
			}
			KOR_wartime_events = yes

			# Recognition decays slowly if nothing is done about it.
			if = {
				limit = { check_variable = { KOR_recognition > 0 } }
				subtract_from_variable = { KOR_recognition = 0.02 }
			}
		}
	}

	on_daily_CHO = {
		effect = {
			if = {
				limit = {
					date > "1936.8.24"
					NOT = { has_country_flag = KOR_trigger_event_fired }
				}
				set_country_flag = KOR_trigger_event_fired
				country_event = { id = korea_triggers.1 days = 1 }
			}
			if = {
				limit = {
					date > "1943.11.30"
					NOT = { has_country_flag = KOR_cairo_fired }
				}
				set_country_flag = KOR_cairo_fired
				country_event = { id = korea_triggers.2 days = 1 }
			}

			KOR_wartime_events = yes
		}
	}

	# ROOT = nation being released, FROM = overlord.
	# When Korea is released by anyone, hand over both core states explicitly.
	# The peace conference has been known to hand over only one of them.
	on_release_as_puppet = {
		effect = {
			if = {
				limit = { tag = KOR }
				every_state = {
					limit = { is_core_of = ROOT }
					transfer_state_to = ROOT
				}
			}
		}
	}
}
