on_actions = {

	on_startup = {
		effect = {
			# One tag. The peninsula and the government sitting on it are the
			# same country in the game's eyes; which of the two it is at any
			# moment is a matter of who holds it and what it calls itself, and
			# both of those are things the mod already tracks.
			KOR = { KOR_common_setup = yes }

			# --- Governorate mode. The peninsula already belongs to Korea in
			# --- the six history/states files, not here, because the country
			# --- selection screen reads history and nothing else. Doing it in
			# --- on_startup meant the peninsula was Japanese while you were
			# --- choosing and then broke away the instant the game began.
			# --- What is left for startup is the diplomacy, which has no map
			# --- to redraw, and the name.
			if = {
				limit = { has_game_rule = { rule = KOR_start_mode option = GOVERNORATE } }

				KOR = {
					add_ideas = KOR_governorate_police
					set_cosmetic_tag = KOR_gg
				}

				JAP = {
					set_autonomy = { target = KOR autonomy_state = autonomy_integrated_puppet }
					# The metropole loses direct use of Korean manpower and
					# industry; the integrated-puppet share plus this spirit
					# is what puts it back.
					add_ideas = JAP_korean_integration
				}

				set_global_flag = KOR_governorate_active
			}

			# --- Releasable-only mode. Hand the peninsula back and leave the
			# --- 1936 map exactly as the base game draws it. Japan's AI then
			# --- sees nothing different, which is the whole point of this
			# --- option; the cost is that Korea cannot be picked at start.
			else = {
				KOR_return_peninsula_to_japan = yes

				# history/countries/KOR - Korea.txt is written for the
				# governorate, because that is the default. A Korea that is
				# only releasable is the exile movement instead, and starts
				# where the base game starts it.
				KOR = {
					set_politics = {
						ruling_party = democratic
						last_election = "1936.1.1"
						election_frequency = 48
						elections_allowed = yes
					}
					set_popularities = {
						democratic = 35
						fascism = 25
						communism = 30
						neutrality = 10
					}
					if = {
						limit = { has_idea = KOR_governorate_police }
						remove_ideas = KOR_governorate_police
					}
				}
			}
		}
	}

	# The flag-defacement incident, and Cairo, are fired from here.
	on_daily_KOR = {
		effect = {
			if = {
				limit = {
					date > "1936.8.24"
					NOT = { has_country_flag = KOR_trigger_event_fired }
					OR = { is_subject = yes owns_state = 525 has_country_flag = KOR_underground_built }
				}
				set_country_flag = KOR_trigger_event_fired
				country_event = { id = korea_triggers.1 days = 1 }
			}
			if = {
				limit = {
					date > "1943.11.30"
					NOT = { has_country_flag = KOR_cairo_fired }
				}
				set_country_flag = KOR_cairo_fired
				country_event = { id = korea_triggers.2 days = 1 }
			}
			KOR_wartime_events = yes
			KOR_refresh_gauges = yes
		}
	}

	# ROOT = nation being released, FROM = overlord.
	# When Korea is released by anyone, hand over both core states explicitly.
	# The peace conference has been known to hand over only one of them.
	on_release_as_puppet = {
		effect = {
			if = {
				limit = { tag = KOR }
				every_state = {
					limit = { is_core_of = ROOT }
					transfer_state_to = ROOT
				}
			}
		}
	}
}
