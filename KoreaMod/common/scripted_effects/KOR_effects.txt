# Chosŏn mod - scripted effects

# ---------------------------------------------------------------------------
# Colonial supply base: 50% of output is siphoned to the overlord. Every
# industry focus claws back 5 percentage points, down to a floor of 20%.
# The number lives in a variable that a dynamic modifier reads every day.
# ---------------------------------------------------------------------------
KOR_setup_leakage = {
	set_variable = { KOR_leak = -0.50 }
	if = { limit = { KOR_difficulty_balanced = yes } set_variable = { KOR_leak = -0.35 } }
	if = { limit = { KOR_difficulty_empowered = yes } set_variable = { KOR_leak = 0 } }
	set_variable = { KOR_leak_floor = -0.20 }
	if = { limit = { KOR_difficulty_empowered = yes } set_variable = { KOR_leak_floor = 0 } }
	set_variable = { KOR_leak_display = KOR_leak }
	multiply_variable = { KOR_leak_display = -100 }
}

KOR_recover_leakage = {
	add_to_variable = { KOR_leak = 0.05 }
	clamp_variable = { var = KOR_leak min = -1 max = KOR_leak_floor }
	set_variable = { KOR_leak_display = KOR_leak }
	multiply_variable = { KOR_leak_display = -100 }
}

# Every tier-1 industry focus calls this. The gate focus only reads the counter.
KOR_count_industry_focus = {
	add_to_variable = { KOR_ind_tier1 = 1 }
	add_to_variable = { KOR_ind_total = 1 }
	KOR_recover_leakage = yes
}

KOR_count_industry_focus_t2 = {
	add_to_variable = { KOR_ind_total = 1 }
	KOR_recover_leakage = yes
}

KOR_count_army_focus = {
	add_to_variable = { KOR_army_focuses = 1 }
}

# ---------------------------------------------------------------------------
# Allied recognition gauge. Range 0-100, capped at 60 until the cap is broken
# by actually putting men on Korean soil.
# ---------------------------------------------------------------------------
KOR_add_recognition = {
	add_to_variable = { KOR_recognition = var:KOR_recognition_delta }
	if = {
		limit = { NOT = { has_country_flag = KOR_recognition_cap_lifted } }
		clamp_variable = { var = KOR_recognition min = 0 max = 60 }
	}
	else = {
		clamp_variable = { var = KOR_recognition min = 0 max = 100 }
	}
}

KOR_lift_recognition_cap = {
	set_country_flag = KOR_recognition_cap_lifted
}

# ---------------------------------------------------------------------------
# Officer corps: three steps down, never gone.
# ---------------------------------------------------------------------------
KOR_officer_stage_2 = {
	if = {
		limit = { has_idea = KOR_officer_shortage_1 }
		swap_ideas = { remove_idea = KOR_officer_shortage_1 add_idea = KOR_officer_shortage_2 }
		set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 8 }
	}
}

KOR_officer_stage_3 = {
	if = {
		limit = { has_idea = KOR_officer_shortage_2 }
		swap_ideas = { remove_idea = KOR_officer_shortage_2 add_idea = KOR_officer_shortage_3 }
	}
	else_if = {
		limit = { has_idea = KOR_officer_shortage_1 }
		swap_ideas = { remove_idea = KOR_officer_shortage_1 add_idea = KOR_officer_shortage_3 }
	}
	set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 999 }
}

# ---------------------------------------------------------------------------
# Equipment naming. A cosmetic tag is the only way to re-name equipment
# inside one country, so each sponsor swap is a tag swap.
# ---------------------------------------------------------------------------
KOR_name_equipment_japanese = { set_cosmetic_tag = KOR_jap }
KOR_name_equipment_china    = { set_cosmetic_tag = KOR_chi }
KOR_name_equipment_usa      = { set_cosmetic_tag = KOR_usa }
KOR_name_equipment_prc      = { set_cosmetic_tag = KOR_csr }
KOR_name_equipment_soviet   = { set_cosmetic_tag = KOR_sov }
KOR_name_equipment_german   = { set_cosmetic_tag = KOR_ger }

# ---------------------------------------------------------------------------
# Left/right balance. Sum is always 100.
# ---------------------------------------------------------------------------
KOR_shift_right = {
	add_to_variable = { KOR_scale_right = var:KOR_scale_delta }
	clamp_variable = { var = KOR_scale_right min = 0 max = 100 }
	set_variable = { KOR_scale_left = 100 }
	subtract_from_variable = { KOR_scale_left = KOR_scale_right }
}

KOR_shift_left = {
	subtract_from_variable = { KOR_scale_right = var:KOR_scale_delta }
	clamp_variable = { var = KOR_scale_right min = 0 max = 100 }
	set_variable = { KOR_scale_left = 100 }
	subtract_from_variable = { KOR_scale_left = KOR_scale_right }
}

# ---------------------------------------------------------------------------
# Sovereignty. Called by every branch that actually ends up running a state.
# ---------------------------------------------------------------------------
KOR_restore_sovereignty = {
	set_country_flag = KOR_sovereignty_restored
	if = { limit = { is_subject = yes } OVERLORD = { end_puppet = ROOT } }
	if = {
		limit = { has_idea = KOR_governorate_police }
		remove_ideas = KOR_governorate_police
	}
	if = {
		limit = { has_idea = KOR_army_hq_veto }
		remove_ideas = KOR_army_hq_veto
	}
	if = {
		limit = { has_idea = KOR_manpower_extraction }
		remove_ideas = KOR_manpower_extraction
	}
	# The peace conference has a bad habit of handing over half a peninsula.
	# Hand it over explicitly instead.
	every_state = {
		limit = { is_core_of = KOR }
		transfer_state_to = ROOT
	}
}

# ---------------------------------------------------------------------------
# Startup. Called for KOR and for CHO, because either of them can be the tag
# the player is actually holding depending on the start-mode game rule.
# Everything the tree, the decisions and the tooltips read must exist on both.
# ---------------------------------------------------------------------------
KOR_common_setup = {
	KOR_setup_leakage = yes

	set_variable = { KOR_recognition = 0 }
	set_variable = { KOR_recognition_delta = 0 }
	set_variable = { KOR_ind_tier1 = 0 }
	set_variable = { KOR_ind_total = 0 }
	set_variable = { KOR_army_focuses = 0 }
	set_variable = { KOR_scale_right = 50 }
	set_variable = { KOR_scale_left = 50 }
	set_variable = { KOR_scale_delta = 0 }
	set_variable = { KOR_cached_infantry = 0 }
	set_variable = { KOR_cached_artillery = 0 }

	add_ideas = {
		KOR_manpower_extraction
		KOR_divided_movement
		KOR_army_hq_veto
		KOR_officer_shortage_1
		KOR_colonial_education
	}

	KOR_apply_difficulty = yes
}

# The tree is identical in all three difficulties. Only the numbers move.
KOR_apply_difficulty = {
	set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 3 }

	if = {
		limit = { KOR_difficulty_balanced = yes }
		swap_ideas = { remove_idea = KOR_officer_shortage_1 add_idea = KOR_officer_shortage_2 }
		set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 8 }
		add_offsite_building = { type = industrial_complex level = 3 }
	}
	if = {
		limit = { KOR_difficulty_empowered = yes }
		remove_ideas = KOR_officer_shortage_1
		remove_ideas = KOR_manpower_extraction
		set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 999 }
		add_offsite_building = { type = industrial_complex level = 8 }
		add_offsite_building = { type = arms_factory level = 4 }
	}
}

# The peninsula, defined once: every state the base game cores to Korea.
# There are six of them - 525 Gyeonggi, 527 Pyongan-Hwanghae, 1028 Hamgyong,
# 1029 Gangwon, 1030 Gyeongsang, 1031 Chungcheong-Jeolla.
KOR_seize_the_peninsula = {
	every_state = {
		limit = { is_core_of = KOR }
		set_state_controller = ROOT
	}
}

KOR_return_peninsula_to_japan = {
	every_state = {
		limit = { is_core_of = KOR }
		transfer_state_to = JAP
		remove_core_of = CHO
	}
}
