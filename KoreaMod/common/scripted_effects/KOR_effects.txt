# Chosŏn mod - scripted effects

# ---------------------------------------------------------------------------
# Colonial supply base: 50% of output is siphoned to the overlord. Every
# industry focus claws back 5 percentage points, down to a floor of 20%.
# The number lives in a variable that a dynamic modifier reads every day.
# ---------------------------------------------------------------------------
KOR_setup_leakage = {
	set_variable = { KOR_leak = -0.50 }
	if = { limit = { KOR_difficulty_balanced = yes } set_variable = { KOR_leak = -0.35 } }
	if = { limit = { KOR_difficulty_empowered = yes } set_variable = { KOR_leak = 0 } }
	set_variable = { KOR_leak_floor = -0.20 }
	if = { limit = { KOR_difficulty_empowered = yes } set_variable = { KOR_leak_floor = 0 } }
	set_variable = { KOR_leak_display = KOR_leak }
	multiply_variable = { KOR_leak_display = -100 }
}

# The spirit and the modifier that reads the variable have to travel
# together, and both have to be safe to grant twice: startup hands them out,
# and the logistics focus asks for them again. add_dynamic_modifier stacks if
# it is called twice, so the guard is not decoration.
KOR_ensure_supply_base = {
	if = {
		limit = { NOT = { has_idea = KOR_colonial_supply_base } }
		add_ideas = KOR_colonial_supply_base
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = KOR_supply_base_leak } } }
		add_dynamic_modifier = { modifier = KOR_supply_base_leak }
	}
}

KOR_recover_leakage = {
	add_to_variable = { KOR_leak = 0.05 }
	clamp_variable = { var = KOR_leak min = -1 max = KOR_leak_floor }
	set_variable = { KOR_leak_display = KOR_leak }
	multiply_variable = { KOR_leak_display = -100 }
}

# The war years push the other way. Same clamp, opposite sign: the floor is a
# floor on how much you can claw back, not on how much can be taken.
KOR_worsen_leakage = {
	# On the empowered rule the leak was set to zero on purpose. Leave it there.
	if = {
		limit = { NOT = { KOR_difficulty_empowered = yes } }
		subtract_from_variable = { KOR_leak = 0.05 }
		clamp_variable = { var = KOR_leak min = -1 max = KOR_leak_floor }
		set_variable = { KOR_leak_display = KOR_leak }
		multiply_variable = { KOR_leak_display = -100 }
	}
}

# Every tier-1 industry focus calls this. The gate focus only reads the counter.
KOR_count_industry_focus = {
	add_to_variable = { KOR_ind_tier1 = 1 }
	add_to_variable = { KOR_ind_total = 1 }
	KOR_recover_leakage = yes
}

KOR_count_industry_focus_t2 = {
	add_to_variable = { KOR_ind_total = 1 }
	KOR_recover_leakage = yes
}

KOR_count_army_focus = {
	add_to_variable = { KOR_army_focuses = 1 }
}

# ---------------------------------------------------------------------------
# Allied recognition gauge. Range 0-100, capped at 60 until the cap is broken
# by actually putting men on Korean soil.
# ---------------------------------------------------------------------------
KOR_add_recognition = {
	add_to_variable = { KOR_recognition = var:KOR_recognition_delta }
	if = {
		limit = { NOT = { has_country_flag = KOR_recognition_cap_lifted } }
		clamp_variable = { var = KOR_recognition min = 0 max = 60 }
	}
	else = {
		clamp_variable = { var = KOR_recognition min = 0 max = 100 }
	}
}

KOR_lift_recognition_cap = {
	set_country_flag = KOR_recognition_cap_lifted
}

# ---------------------------------------------------------------------------
# Officer corps: three steps down, never gone.
# ---------------------------------------------------------------------------
KOR_officer_stage_2 = {
	if = {
		limit = { has_idea = KOR_officer_shortage_1 }
		swap_ideas = { remove_idea = KOR_officer_shortage_1 add_idea = KOR_officer_shortage_2 }
		set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 8 }
	}
}

KOR_officer_stage_3 = {
	if = {
		limit = { has_idea = KOR_officer_shortage_2 }
		swap_ideas = { remove_idea = KOR_officer_shortage_2 add_idea = KOR_officer_shortage_3 }
	}
	else_if = {
		limit = { has_idea = KOR_officer_shortage_1 }
		swap_ideas = { remove_idea = KOR_officer_shortage_1 add_idea = KOR_officer_shortage_3 }
	}
	set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 999 }
}

# ---------------------------------------------------------------------------
# Equipment naming. A cosmetic tag is the only way to re-name equipment
# inside one country, so each sponsor swap is a tag swap.
#
# There is only one cosmetic tag slot, so a country that has also re-named
# itself cannot hold a separate naming tag: whichever effect runs last wins
# and the other one's names are lost. Where the two collide - the People's
# Republic and the restored Yi state - the naming tag and the country name
# live on the same tag, and its equipment list is a copy of the sponsor's
# in localisation/*/KOR_equipment_l_*.yml.
# ---------------------------------------------------------------------------
# The governorate names itself. KOR_jap stays "조선" because the People's
# Republic uses it for its own captured-Japanese arsenal, and the committees
# are not the Government-General.
KOR_name_equipment_governorate = { set_cosmetic_tag = KOR_gg }

KOR_name_equipment_japanese = { set_cosmetic_tag = KOR_jap }
KOR_name_equipment_china    = { set_cosmetic_tag = KOR_chi }
KOR_name_equipment_usa      = { set_cosmetic_tag = KOR_usa }
KOR_name_equipment_prc      = { set_cosmetic_tag = KOR_csr }
KOR_name_equipment_soviet   = { set_cosmetic_tag = KOR_sov }

# The German mission is only reachable from YIH_reproclaim_empire, so the
# state has already been re-named by korea_politics.6 when this runs. Keep
# that name and change only the arsenal.
KOR_name_equipment_german = {
	if = {
		limit = { has_country_flag = YIH_yi_un_chosen }
		set_cosmetic_tag = KOR_dominion_ger
	}
	else_if = {
		limit = { has_country_flag = YIH_empire_proclaimed }
		set_cosmetic_tag = KOR_empire_ger
	}
	else = {
		set_cosmetic_tag = KOR_ger
	}
}

# ---------------------------------------------------------------------------
# The war years. Dated events that happen to the peninsula whichever branch
# is being played, plus the one event CHO_autonomous_assembly promises in red
# text. Called from both daily on_actions so it runs on whichever tag is held.
# ---------------------------------------------------------------------------
KOR_wartime_events = {
	if = {
		limit = {
			date > "1937.7.6"
			NOT = { has_country_flag = KOR_rear_base_fired }
		}
		set_country_flag = KOR_rear_base_fired
		country_event = { id = korea_triggers.3 days = 1 }
	}
	if = {
		limit = {
			date > "1941.9.30"
			NOT = { has_country_flag = KOR_supung_fired }
		}
		set_country_flag = KOR_supung_fired
		country_event = { id = korea_industry.2 days = 1 }
	}
	if = {
		limit = {
			date > "1941.12.8"
			NOT = { has_country_flag = KOR_declaration_fired }
		}
		set_country_flag = KOR_declaration_fired
		country_event = { id = korea_triggers.4 days = 1 }
	}
	if = {
		limit = {
			date > "1942.9.30"
			NOT = { has_country_flag = KOR_language_society_fired }
		}
		set_country_flag = KOR_language_society_fired
		country_event = { id = korea_triggers.5 days = 1 }
	}
	if = {
		limit = {
			date > "1944.3.31"
			NOT = { has_country_flag = KOR_conscription_fired }
		}
		set_country_flag = KOR_conscription_fired
		country_event = { id = korea_triggers.6 days = 1 }
	}
	# The assembly's own tooltip says this government does not outlive Japan.
	if = {
		limit = {
			has_country_flag = CHO_dominion_declared
			NOT = { has_country_flag = CHO_dominion_fallen }
			OR = {
				JAP = { has_capitulated = yes }
				JAP = { surrender_progress > 0.85 }
			}
		}
		set_country_flag = CHO_dominion_fallen
		country_event = { id = korea_politics.7 days = 1 }
	}
}

# ---------------------------------------------------------------------------
# Left/right balance. Sum is always 100.
# ---------------------------------------------------------------------------
KOR_shift_right = {
	add_to_variable = { KOR_scale_right = var:KOR_scale_delta }
	clamp_variable = { var = KOR_scale_right min = 0 max = 100 }
	set_variable = { KOR_scale_left = 100 }
	subtract_from_variable = { KOR_scale_left = KOR_scale_right }
}

KOR_shift_left = {
	subtract_from_variable = { KOR_scale_right = var:KOR_scale_delta }
	clamp_variable = { var = KOR_scale_right min = 0 max = 100 }
	set_variable = { KOR_scale_left = 100 }
	subtract_from_variable = { KOR_scale_left = KOR_scale_right }
}

# ---------------------------------------------------------------------------
# Sovereignty. Called by every branch that actually ends up running a state.
# ---------------------------------------------------------------------------
KOR_restore_sovereignty = {
	set_country_flag = KOR_sovereignty_restored
	if = { limit = { is_subject = yes } OVERLORD = { end_puppet = ROOT } }
	if = {
		limit = { has_idea = KOR_governorate_police }
		remove_ideas = KOR_governorate_police
	}
	if = {
		limit = { has_idea = KOR_army_hq_veto }
		remove_ideas = KOR_army_hq_veto
	}
	if = {
		limit = { has_idea = KOR_manpower_extraction }
		remove_ideas = KOR_manpower_extraction
	}
	# The peace conference has a bad habit of handing over half a peninsula.
	# Hand it over explicitly instead.
	every_state = {
		limit = { is_core_of = KOR }
		transfer_state_to = ROOT
	}
}

# ---------------------------------------------------------------------------
# Startup. Called for KOR and for CHO, because either of them can be the tag
# the player is actually holding depending on the start-mode game rule.
# Everything the tree, the decisions and the tooltips read must exist on both.
# ---------------------------------------------------------------------------
# ---------------------------------------------------------------------------
# The gauges, once a day. A number the player can move is only a system if
# something reads it while the game is running; recognition and the left-right
# scale were being read at decision time and nowhere else. These lines are what
# the two dynamic modifiers below consume, so the gauges press on the game the
# way the leak rate already did.
# ---------------------------------------------------------------------------
KOR_refresh_gauges = {
	# Recognition decays if nothing is being done about it. Both tags, so the
	# governorate cannot bank standing it is not working for.
	if = {
		limit = { check_variable = { KOR_recognition > 0 } }
		subtract_from_variable = { KOR_recognition = 0.02 }
	}

	# 0-100 standing -> political weight and the war support that comes with
	# being a government somebody has heard of.
	set_variable = { KOR_recog_weight = KOR_recognition }
	divide_variable = { KOR_recog_weight = 400 }
	set_variable = { KOR_recog_support = KOR_recognition }
	divide_variable = { KOR_recog_support = 1000 }

	# The scale is two numbers summing to 100, so how far it leans is the
	# distance from the middle. Leaning buys zeal and costs the united front.
	set_variable = { KOR_scale_left = 100 }
	subtract_from_variable = { KOR_scale_left = KOR_scale_right }
	set_variable = { KOR_scale_gap = KOR_scale_right }
	subtract_from_variable = { KOR_scale_gap = 50 }
	if = {
		limit = { check_variable = { KOR_scale_gap < 0 } }
		multiply_variable = { KOR_scale_gap = -1 }
	}
	set_variable = { KOR_front_cohesion = KOR_scale_gap }
	divide_variable = { KOR_front_cohesion = -200 }
	set_variable = { KOR_front_zeal = KOR_scale_gap }
	divide_variable = { KOR_front_zeal = 500 }
}

KOR_common_setup = {
	KOR_setup_leakage = yes

	set_variable = { KOR_recognition = 0 }
	set_variable = { KOR_recognition_delta = 0 }
	set_variable = { KOR_ind_tier1 = 0 }
	set_variable = { KOR_ind_total = 0 }
	set_variable = { KOR_army_focuses = 0 }
	set_variable = { KOR_scale_right = 50 }
	set_variable = { KOR_scale_left = 50 }
	set_variable = { KOR_scale_delta = 0 }
	set_variable = { KOR_cached_infantry = 0 }
	set_variable = { KOR_cached_artillery = 0 }

	add_ideas = {
		KOR_manpower_extraction
		KOR_divided_movement
		KOR_army_hq_veto
		KOR_officer_shortage_1
		KOR_colonial_education
	}

	# The peninsula is a supply base on 1 January 1936, not from whenever the
	# player happens to take the logistics focus. Until this moved here the
	# spirit arrived with that focus, so KOR_setup_leakage was computing a
	# leak rate that nothing read and the industry focuses were clawing back
	# points off a modifier that was not attached yet.
	KOR_ensure_supply_base = yes

	KOR_refresh_gauges = yes
	add_dynamic_modifier = { modifier = KOR_allied_standing }
	add_dynamic_modifier = { modifier = KOR_united_front_strain }

	KOR_apply_difficulty = yes
}

# The tree is identical in all three difficulties. Only the numbers move.
KOR_apply_difficulty = {
	set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 3 }

	if = {
		limit = { KOR_difficulty_balanced = yes }
		swap_ideas = { remove_idea = KOR_officer_shortage_1 add_idea = KOR_officer_shortage_2 }
		set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 8 }
		add_offsite_building = { type = industrial_complex level = 3 }
	}
	if = {
		limit = { KOR_difficulty_empowered = yes }
		remove_ideas = KOR_officer_shortage_1
		remove_ideas = KOR_manpower_extraction
		set_division_template_cap = { division_template = "Daehanminguk Yuk-gun" division_cap = 999 }
		add_offsite_building = { type = industrial_complex level = 8 }
		add_offsite_building = { type = arms_factory level = 4 }
	}
}

# The peninsula, defined once: every state the base game cores to Korea.
# There are six of them - 525 Gyeonggi, 527 Pyongan-Hwanghae, 1028 Hamgyong,
# 1029 Gangwon, 1030 Gyeongsang, 1031 Chungcheong-Jeolla.
KOR_seize_the_peninsula = {
	every_state = {
		limit = { is_core_of = KOR }
		set_state_controller = ROOT
	}
}

KOR_return_peninsula_to_japan = {
	every_state = {
		limit = { is_core_of = KOR }
		transfer_state_to = JAP
		remove_core_of = CHO
	}
}
