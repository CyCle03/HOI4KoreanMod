# Written for the governorate, because GOVERNORATE is the default start mode
# and in it this tag *is* the Government-General. The LIBERATION branch of
# on_startup puts the politics back to the exile movement's, which is where
# the base game leaves Korea.

	infantry_weapons = 1
	tech_support = 1
	tech_recon = 1
	tech_engineers = 1
	gw_artillery = 1
	tech_trucks = 1
	basic_train = 1
}

set_grand_doctrine = mass_assault

set_research_slots = 2
set_convoys = 10
set_stability = 0.60
set_war_support = 0.20

set_politics = {
	ruling_party = neutrality
	last_election = "1936.1.1"
	election_frequency = 48
	elections_allowed = no
}

# Not 100/0/0/0. The exile movements are small and illegal, not imaginary,
# and KOR_divided_movement pushes both of these upward every day the police
# state is not holding them down. Starting them at zero made the political
# tree's five branches grow out of nothing.
set_popularities = {
	democratic = 5
	fascism = 0
	communism = 5
	neutrality = 90
}

	promote_leader = yes

		ideology = oligarchism
		traits = { JAP_governor_general_of_korea }
		expire = "1965.1.1.1"

	}
